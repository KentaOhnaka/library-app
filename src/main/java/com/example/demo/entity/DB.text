-- =================================================================
-- テーブルが存在する場合、一度綺麗に更地にするための呪文（開発時のみ）
-- =================================================================
DROP TABLE IF EXISTS RESERVE;
DROP TABLE IF EXISTS STOCK; -- ※拙者が提案した在庫テーブル。採用する場合に使う。
DROP TABLE IF EXISTS NOTICE;
DROP TABLE IF EXISTS BOOK;
DROP TABLE IF EXISTS AUTHOR;
DROP TABLE IF EXISTS LIBRARY;
DROP TABLE IF EXISTS WARD;
DROP TABLE IF EXISTS USERS;


-- =================================================================
-- 1. 区画の巻 (WARD)
-- =================================================================
CREATE TABLE WARD (
    WARD_CODE VARCHAR(10) NOT NULL,            -- 区コード (主キー)
    WARD_NAME VARCHAR(50) NOT NULL,            -- 区名
    WARD_COORDINATES VARCHAR(255),             -- 区の座標情報
    PRIMARY KEY (WARD_CODE)
);
COMMENT ON TABLE WARD IS '区画マスタ';
COMMENT ON COLUMN WARD.WARD_CODE IS '区コード';
COMMENT ON COLUMN WARD.WARD_NAME IS '区名';
COMMENT ON COLUMN WARD.WARD_COORDINATES IS '区の座標情報';


-- =================================================================
-- 2. 図書館の巻 (LIBRARY)
-- =================================================================
CREATE TABLE LIBRARY (
    LIBRARY_CODE VARCHAR(10) NOT NULL,         -- 図書館コード (主キー)
    LIBRARY_NAME VARCHAR(100) NOT NULL,        -- 図書館名
    LIBRARY_TEL VARCHAR(20),                   -- 電話番号
    LIBRARY_FAX VARCHAR(20),                   -- FAX番号
    LIBRARY_ADDRESS VARCHAR(255),              -- 住所
    OPENING_HOURS VARCHAR(255),                -- 開館時間
    WARD_CODE VARCHAR(10),                     -- 区コード (WARDテーブルへの参照)
    PRIMARY KEY (LIBRARY_CODE),
    FOREIGN KEY (WARD_CODE) REFERENCES WARD(WARD_CODE)
);
COMMENT ON TABLE LIBRARY IS '図書館マスタ';
COMMENT ON COLUMN LIBRARY.LIBRARY_CODE IS '図書館コード';
-- ... (以下、他のカラムへのコメントも同様)


-- =================================================================
-- 3. 著者の巻 (AUTHOR)
-- =================================================================
CREATE TABLE AUTHOR (
    AUTHOR_CODE VARCHAR(10) NOT NULL,          -- 著者コード (主キー)
    AUTHOR_NAME VARCHAR(100) NOT NULL,         -- 著者名
    AUTHOR_MAIL VARCHAR(255),                  -- 著者メール
    AUTHOR_HOMEPAGE VARCHAR(255),              -- 著者HP
    AUTHOR_BELONG VARCHAR(100),                -- 著者所属
    PRIMARY KEY (AUTHOR_CODE)
);
COMMENT ON TABLE AUTHOR IS '著者マスタ';


-- =================================================================
-- 4. 書物の巻 (BOOK)
-- =================================================================
CREATE TABLE BOOK (
    ISBN VARCHAR(20) NOT NULL,                 -- ISBNコード (主キー)
    BOOK_NAME VARCHAR(255) NOT NULL,           -- 書籍名
    AUTHOR_CODE VARCHAR(10) NOT NULL,          -- 著者コード (AUTHORテーブルへの参照)
    PUBLISH_DATE DATE,                         -- 出版日
    PUBLISH_COMPANY VARCHAR(100),              -- 出版社
    BOOK_SUMMARY TEXT,                         -- 書籍概要
    PRIMARY KEY (ISBN),
    FOREIGN KEY (AUTHOR_CODE) REFERENCES AUTHOR(AUTHOR_CODE)
);
COMMENT ON TABLE BOOK IS '書籍マスタ';


-- =================================================================
-- 5. 利用者の巻 (USERS)
-- =================================================================
CREATE TABLE USERS (
    USER_ID VARCHAR(50) NOT NULL,              -- ユーザーID (主キー)
    USER_PASSWORD VARCHAR(255) NOT NULL,       -- パスワード (ハッシュ化して保存すること)
    USER_NAME VARCHAR(100) NOT NULL,           -- ユーザー名
    USER_ADMIN_FLG BOOLEAN NOT NULL DEFAULT FALSE, -- 管理者フラグ (true:管理者)
    PRIMARY KEY (USER_ID)
);
COMMENT ON TABLE USERS IS '利用者マスタ';


-- =================================================================
-- 6. 予約の巻 (RESERVE)
-- =================================================================
CREATE TABLE RESERVE (
    RESERVE_ID SERIAL NOT NULL,                -- 予約ID (主キー, 自動採番)
    USER_ID VARCHAR(50) NOT NULL,              -- ユーザーID (USERSテーブルへの参照)
    ISBN VARCHAR(20) NOT NULL,                 -- ISBNコード (BOOKテーブルへの参照)
    LIBRARY_CODE VARCHAR(10) NOT NULL,         -- 受取図書館コード (LIBRARYテーブルへの参照)
    RESERVE_DATE TIMESTAMP NOT NULL,           -- 予約日時
    RESERVE_NOTICE_METHOD VARCHAR(50),         -- 予約確保の通知方法
    RESERVE_MEMO TEXT,                         -- メモ
    PRIMARY KEY (RESERVE_ID),
    FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID),
    FOREIGN KEY (ISBN) REFERENCES BOOK(ISBN),
    FOREIGN KEY (LIBRARY_CODE) REFERENCES LIBRARY(LIBRARY_CODE)
);
COMMENT ON TABLE RESERVE IS '予約テーブル';


-- =================================================================
-- 7. お知らせの巻 (NOTICE)
-- =================================================================
CREATE TABLE NOTICE (
    NOTICE_CODE SERIAL NOT NULL,               -- お知らせコード (主キー, 自動採番)
    NOTICE_TITLE VARCHAR(255) NOT NULL,        -- お知らせタイトル
    LIBRARY_CODE VARCHAR(10),                  -- お知らせを発信した図書館 (NULL許容は全館共通のお知らせ)
    NOTICE_DATE DATE NOT NULL,                 -- お知らせ日付
    NOTICE_CONTENTS TEXT,                      -- お知らせ内容
    PRIMARY KEY (NOTICE_CODE),
    FOREIGN KEY (LIBRARY_CODE) REFERENCES LIBRARY(LIBRARY_CODE)
);
COMMENT ON TABLE NOTICE IS 'お知らせテーブル';
